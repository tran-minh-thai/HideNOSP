  /** Positions carrying a node of P's Nettree that lies on a complete occurrence (root level to leaf level). */
  static TreeSet<Integer> onOccurrences(Pattern p, EventSequence seq, GapConstraint gap) {
    Nettree t = Nettree.build(p, seq, gap); int n = t.nodeCount();
    boolean[] fromRoot = new boolean[n], toLeaf = new boolean[n];
    // Nodes are numbered level by level, so one pass in each direction settles reachability.
    for (int v = 0; v < n; v++) { if (t.isSource(v)) fromRoot[v] = true; if (fromRoot[v]) for (int u : t.out(v)) fromRoot[u] = true; }
    for (int v = n - 1; v >= 0; v--) { if (t.isSink(v)) toLeaf[v] = true; if (toLeaf[v]) for (int u : t.in(v)) toLeaf[u] = true; }
    TreeSet<Integer> out = new TreeSet<>(); for (int v = 0; v < n; v++) if (fromRoot[v] && toLeaf[v]) out.add(t.position(v));
    return out;
  }
  /**
   * Marginal damage at the current solution, per masked sequence and watched pattern present there:
   * outside the masks, 1 on positions carrying a vital node of Q in the masked sequence; on a mask,
   * the support of Q regained by unmasking that position alone.
   */
  static Map<Integer, Map<Pattern, double[]>> marginal(SequenceDatabase db, GapConstraint gap, Map<Integer, Set<Integer>> masked, Map<Integer, Map<Pattern, Integer>> supIn) {
    Map<Integer, Map<Pattern, double[]>> out = new HashMap<>();
    for (Map.Entry<Integer, Set<Integer>> e : masked.entrySet()) {
      if (e.getValue().isEmpty()) continue;
      EventSequence orig = db.get(e.getKey()), after = orig.mask(e.getValue());
      NettreeCutIndex idx = new NettreeCutIndex(after, gap);
      Map<Pattern, double[]> m = new HashMap<>();
      for (Pattern q : supIn.get(e.getKey()).keySet()) {
        double[] a = new double[orig.length()];
        int id = idx.add(q, false); int now = idx.supNO(id);
        for (int pos : idx.vitalPositions(id)) a[pos] = 1;
        for (int i : e.getValue()) { Set<Integer> rest = new HashSet<>(e.getValue()); rest.remove(i); a[i] = SupportNO.of(q, orig.mask(rest), gap) - now; }
        m.put(q, a);
      }
      out.put(e.getKey(), m);
    }
    return out;
  }
  static Map<Integer, List<Integer>> toLists(Map<Integer, Set<Integer>> m) { Map<Integer, List<Integer>> out = new HashMap<>(); for (Map.Entry<Integer, Set<Integer>> e : m.entrySet()) out.put(e.getKey(), new ArrayList<>(e.getValue())); return out; }
  static long binomial(int n, int k) { long b = 1; for (int i = 0; i < k; i++) { b = b * (n - i) / (i + 1); if (b > Long.MAX_VALUE / 64) return Long.MAX_VALUE / 64; } return b; }
  static Map<Pattern, Long> drops(SequenceDatabase db, GapConstraint gap, Map<Integer, Set<Integer>> masked, Map<Integer, Map<Pattern, Integer>> supIn) {
    Map<Pattern, Long> drop = new HashMap<>();
    for (Map.Entry<Integer, Set<Integer>> e : masked.entrySet()) {
      if (e.getValue().isEmpty()) continue;
      EventSequence after = db.get(e.getKey()).mask(e.getValue());
      for (Map.Entry<Pattern, Integer> sq : supIn.get(e.getKey()).entrySet()) { int now = SupportNO.of(sq.getKey(), after, gap); if (now != sq.getValue()) drop.merge(sq.getKey(), (long) (sq.getValue() - now), Long::sum); }
    }
    return drop;
  }

import hidenosp.core.*; import hidenosp.hide.*; import hidenosp.metrics.*; import hidenosp.index.NettreeCutIndex; import java.util.*;
/** Search: random instances on which the local search WITHOUT the mask-count check of allocation moves changes |R|. */
public class FindSizeChange {
  public static void main(String[] a) {
    long seed = Long.parseLong(a[0]); int trials = Integer.parseInt(a[1]); Random random = new Random(seed); int tried = 0, found = 0;
    for (int trial = 0; trial < trials && found < 5; trial++) {
      int alphabet = 2 + random.nextInt(2);
      List<EventSequence> list = new ArrayList<>(); int sequences = 3 + random.nextInt(6);
      for (int s = 0; s < sequences; s++) { int[] ev = new int[4 + random.nextInt(10)]; for (int i = 0; i < ev.length; i++) ev[i] = random.nextInt(alphabet); list.add(new EventSequence(s, ev)); }
      SequenceDatabase db = new SequenceDatabase(list);
      GapConstraint gap = GapConstraint.of(0, random.nextInt(3));
      int len = 2 + random.nextInt(2); int[] items = new int[len]; for (int j = 0; j < len; j++) items[j] = random.nextInt(alphabet);
      Pattern p = new Pattern(items); if (p.mu() < 2) continue;
      long support = SupportNO.of(p, db, gap); if (support < 3) continue;
      int sigma = (int) Math.max(2, support - 1 - random.nextInt(4));
      HideProblem problem = new HideProblem(db, List.of(p), gap, HideProblem.Mode.HIDE_DB, sigma, 0.5, 3);
      Metrics.Context context = Metrics.Context.of(db, gap, 3, sigma);
      Map<Pattern, Long> f = context.frequent(); Set<Pattern> closure = FrequentPatterns.closure(List.of(p), f.keySet());
      List<Pattern> band = FrequentPatterns.nearThreshold(f, sigma, 0.5, closure); if (band.isEmpty()) continue;
      Map<Pattern, Long> watched = new LinkedHashMap<>(); for (Map.Entry<Pattern, Long> e : f.entrySet()) if (!closure.contains(e.getKey())) watched.put(e.getKey(), e.getValue());
      Map<Integer, Map<Pattern, int[]>> vital = new LinkedHashMap<>();
      for (int s = 0; s < db.size(); s++) { if (SupportNO.of(p, db.get(s), gap) == 0) continue; NettreeCutIndex idx = new NettreeCutIndex(db.get(s), gap); Map<Pattern, int[]> h = new LinkedHashMap<>(); for (Pattern q : watched.keySet()) { int id = idx.add(q, false); if (idx.supNO(id) > 0) h.put(q, idx.vitalPositions(id)); } vital.put(s, h); }
      Metrics.Context.Session session = context.session(problem);
      Sanitization start = new CutHide(CutHideConfig.standard()).withNearThreshold(band, FrequentPatterns.weightFactors(band, f, sigma)).run(problem);
      tried++;
      MaskLocalSearch.Result r = new MaskLocalSearch(problem, watched, sigma, vital, session::drops, 50, 30).run(start);
      int masks = 0; for (Set<Integer> m : r.masks().values()) masks += m.size();
      if (masks != start.maskCount()) { found++; System.out.println("trial " + trial + ": |R| " + start.maskCount() + " -> " + masks + ", P=" + p + " gap " + gap + " sigma " + sigma + " db " + list); }
    }
    System.out.println("instances tried " + tried + ", with |R| changed " + found);
  }
}
