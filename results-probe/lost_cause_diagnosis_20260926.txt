import hidenosp.core.*; import hidenosp.io.*; import hidenosp.metrics.*; import hidenosp.hide.*; import hidenosp.exp.SensitiveSampling; import hidenosp.index.NettreeCutIndex; import java.util.*; import java.nio.file.*;
/** Probe: why are patterns lost? Every lost pattern of CUT-HIDE (standard) is classified as out of the band, under-seen by the vital count, or seen. */
public class LostDiagnosis {
  public static void main(String[] a) {
    String file = a[0]; int sigma = Integer.parseInt(a[1]); GapConstraint gap = GapConstraint.of(0, Integer.parseInt(a[2])); boolean repeated = a[3].equals("repeated");
    SequenceDatabase db = DatasetReader.read(Paths.get(file), DatasetReader.Format.SPMF).database();
    Metrics.Context context = Metrics.Context.of(db, gap, 4, sigma);
    Map<Pattern, Long> f = context.frequent();
    SensitiveSampling.Result sampling = SensitiveSampling.draw(f, 1, 4, 2026, repeated);
    int out = 0, under = 0, seen = 0, lostTotal = 0, sets = 0, outNearDoubled = 0;
    long underGap = 0;
    for (SensitiveSampling.Draw d : sampling.draws()) {
      List<Pattern> sens = d.patterns();
      HideProblem problem = new HideProblem(db, sens, gap, HideProblem.Mode.HIDE_DB, sigma, 0.5, 4);
      Set<Pattern> closure = FrequentPatterns.closure(sens, f.keySet());
      List<Pattern> band = FrequentPatterns.nearThreshold(f, sigma, 0.5, closure);
      double[] factor = FrequentPatterns.weightFactors(band, f, sigma);
      Sanitization san = new CutHide(CutHideConfig.standard()).withNearThreshold(band, factor).run(problem);
      List<Pattern> watched = new ArrayList<>(); for (Pattern q : f.keySet()) if (!closure.contains(q)) watched.add(q);
      Map<Pattern, Long> drop = new HashMap<>(); Map<Pattern, Long> vitalHit = new HashMap<>();
      for (Map.Entry<Integer, Set<Integer>> e : san.masked().entrySet()) {
        if (e.getValue().isEmpty()) continue;
        EventSequence s = db.get(e.getKey()), after = s.mask(e.getValue());
        NettreeCutIndex idx = null;
        for (Pattern q : watched) {
          int before = SupportNO.of(q, s, gap); if (before == 0) continue;
          int now = SupportNO.of(q, after, gap);
          if (now != before) drop.merge(q, (long) (before - now), Long::sum);
          if (idx == null) idx = new NettreeCutIndex(s, gap);
          int id = idx.add(q, false);
          long hits = 0; for (int p : idx.vitalPositions(id)) if (e.getValue().contains(p)) hits++;
          if (hits > 0) vitalHit.merge(q, hits, Long::sum);
        }
      }
      for (Pattern q : watched) {
        long sup = f.get(q), dq = drop.getOrDefault(q, 0L);
        if (sup - dq >= sigma) continue;
        lostTotal++;
        long vq = vitalHit.getOrDefault(q, 0L);
        if (sup >= 1.5 * sigma) { out++; if (sup < 3.0 * sigma) outNearDoubled++; }
        else if (dq > vq) { under++; underGap += dq - vq; }
        else seen++;
      }
      sets++;
    }
    System.out.printf(Locale.ROOT, "%s sigma=%d %s %s: sets %d, lost patterns %d = out of band %d (of which below 3 sigma %d) + under-seen %d (drop above the vital count by %d in all) + seen %d%n",
      Paths.get(file).getFileName(), sigma, gap, a[3], sets, lostTotal, out, outNearDoubled, under, underGap, seen);
  }
}
