/**
 * Probe: a lower bound on the least Lost among the minimum-mask hiding sets, to tell whether a set
 * where CURRENT and RW_VITAL lose the same number of patterns is already optimal.
 *
 * Bound. For a watched pattern Q, a sequence s and any minimum cut C_s of Q's Nettree in s, masking
 * the positions X lowers sup_NO(Q, s) by at least |C_s ∩ nodes(X)|: C_s minus the masked nodes still
 * cuts every surviving occurrence. So for every minimum-mask hiding set X, the total drop D_Q(X) is at
 * least sum_s |C_s ∩ nodes(X_s)|, an additive weight whose least value over the minimum-mask sets
 * the exact program computes. If that least value reaches m_Q = sup(Q) - sigma + 1, Q is lost in
 * every minimum-mask hiding set ("doomed"), and the number of doomed patterns is a lower bound on
 * the optimum. Two cuts per sequence are tried: the source-side minimum cut, and a minimum cut that
 * prefers nodes at positions where P's masks can go. The larger least value is kept.
 */

            } else {
              long[] cost = new long[nq.nodeCount()];
              for (int v = 0; v < cost.length; v++) { int pos = nq.position(v); cost[v] = pVital.get(s).contains(pos) ? 0 : pPos.get(s).contains(pos) ? 1 : 2; }
              cutNodes = NettreeFlow.minWeightMinCut(nq, cost, null).nodes();
            }
            if (cutNodes.length != supIn.get(s).get(q)) throw new IllegalStateException("cut size " + cutNodes.length + " != sup " + supIn.get(s).get(q));
            for (int v : cutNodes) w[s][nq.position(v)] += 1.0;
          }
          Sanitization san = new CutHide(CutHideConfig.uniformWeights()).withExplicitWeights(w).run(problem);
          long least = 0; for (Map.Entry<Integer, Set<Integer>> e : san.masked().entrySet()) for (int i : e.getValue()) least += Math.round(w[e.getKey()][i]);
          if (san.maskCount() != cur.maskCount()) throw new IllegalStateException("|R| " + san.maskCount() + " != " + cur.maskCount());
          best = Math.max(best, least);
        }
        checked++;
        if (best >= m) doomed++;
      }
      if (doomed > rwLost || doomed > curLost) throw new IllegalStateException("bound above a solution: " + doomed);
      rows.printf(Locale.ROOT, "%s;%d;%s;%d;%d;%d;%d%n", config, sets, p, cur.maskCount(), curLost, rwLost, doomed);
      rows.flush();
      sumDoomed += doomed; sumCur += curLost; sumRw += rwLost; if (curLost == doomed) certCur++; if (rwLost == doomed) certRw++;
      sets++;
    }
    System.out.printf(Locale.ROOT, "%s sigma=%d %s %s: %d sets; lost CURRENT %d, RW_VITAL %d, doomed (lower bound) %d over %d patterns checked; certified optimal: CURRENT %d sets, RW_VITAL %d sets%n",
      Paths.get(file).getFileName(), sigma, gap, a[3], sets, sumCur, sumRw, sumDoomed, checked, certCur, certRw);
    rows.close();
  }
  static Map<Pattern, Long> drops(SequenceDatabase db, GapConstraint gap, Sanitization san, Map<Integer, Map<Pattern, Integer>> supIn) {
    Map<Pattern, Long> drop = new HashMap<>();
    for (Map.Entry<Integer, Set<Integer>> e : san.masked().entrySet()) {
      if (e.getValue().isEmpty()) continue;
      EventSequence after = db.get(e.getKey()).mask(e.getValue());
      for (Map.Entry<Pattern, Integer> sq : supIn.get(e.getKey()).entrySet()) {
        int now = SupportNO.of(sq.getKey(), after, gap);
        if (now != sq.getValue()) drop.merge(sq.getKey(), (long) (sq.getValue() - now), Long::sum);
      }
    }
    return drop;
  }
