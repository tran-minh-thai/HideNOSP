/**
 * Probe: the exact least Lost among the minimum-mask hiding sets, on small instances cut from the
 * real data, against CURRENT (the method as it stands) and RW_VITAL (re-weighting on exact drops);
 * and a check of the min-cut lower bound (every pattern it declares doomed must be lost in every
 * minimum-mask set). Version 3 adds RW_LOCAL: re-weighting on the marginal damage at the current
 * solution instead of the vitality in the unmasked data. For a position outside the current set: 1
 * if it carries a vital node of Q in the masked sequence; for a position inside: the support of Q
 * regained by unmasking it alone. Sequences without masks keep their vital positions. Iteration 0
 * is CURRENT's weight, as for RW_VITAL; the best Lost is kept.
 *
 * Instance: for a sensitive pattern P of the probe sample, K windows of W consecutive events cut
 * from real sequences carrying P, each window holding an occurrence of P and at most MAX_POSITIONS
 * positions on complete occurrences of P, plus K windows from sequences without P, drawn with a
 * fixed seed (a sequence shorter than W is kept whole). sigma is set so that the support of P must
 * fall by r = 2, 3 or 4; F_sigma with l_max = 4. Every set of |R|* positions among the positions on
 * complete occurrences of P is tried; no other position can lower the support of P.

        for (int s : supIn.keySet()) { Nettree np = Nettree.build(p, db.get(s), gap); Set<Integer> vp = new HashSet<>(); for (int i : new NettreeFlow(np).vitalPositions()) vp.add(i); Set<Integer> pp = new HashSet<>(); for (int v = 0; v < np.nodeCount(); v++) pp.add(np.position(v)); pVital.put(s, vp); pPos.put(s, pp); }
        int lbDoomed = 0;
        for (Pattern q : watched) {
          long m = f.get(q) - sigma + 1, best = 0;
          for (int variant = 0; variant < 2 && best < m; variant++) {
            double[][] w = new double[N][];
            for (int s = 0; s < N; s++) {
              w[s] = new double[db.get(s).length()];
              if (!supIn.containsKey(s) || !supIn.get(s).containsKey(q)) continue;
              Nettree nq = Nettree.build(q, db.get(s), gap); int[] cutNodes;
              if (variant == 0) { boolean[] c = new NettreeFlow(nq).minCut(); List<Integer> l = new ArrayList<>(); for (int v = 0; v < c.length; v++) if (c[v]) l.add(v); cutNodes = l.stream().mapToInt(Integer::intValue).toArray(); }
              else { long[] cost = new long[nq.nodeCount()]; for (int v = 0; v < cost.length; v++) { int pos = nq.position(v); cost[v] = pVital.get(s).contains(pos) ? 0 : pPos.get(s).contains(pos) ? 1 : 2; } cutNodes = NettreeFlow.minWeightMinCut(nq, cost, null).nodes(); }
              for (int v : cutNodes) w[s][nq.position(v)] += 1.0;
            }
            Sanitization san = new CutHide(CutHideConfig.uniformWeights()).withExplicitWeights(w).run(problem);
            long least = 0; for (Map.Entry<Integer, Set<Integer>> e : san.masked().entrySet()) for (int i : e.getValue()) least += Math.round(w[e.getKey()][i]);
            best = Math.max(best, least);
          }
          if (best >= m) { lbDoomed++; if (!trueDoomed.contains(q)) lbSoundFail++; }
        }
        rows.printf(Locale.ROOT, "%s;%d;%d;%d;%d;%d;%d;%d;%d;%d;%d;%d;%d%n", where, sigma, rstar, cand.size(), hiding, watched.size(), curLost, rwLost, rwLocal, opt, worst, lbDoomed, trueDoomed.size());
        rows.flush();
        instances++; sumCur += curLost; sumRw += rwLost; sumLocal += rwLocal; if (rwLocal == opt) localOpt++; if (Math.min(rwLost, rwLocal) == opt) bothOpt++; sumBoth += Math.min(rwLost, rwLocal); sumOpt += opt; sumWorst += worst; sumLb += lbDoomed; sumTrueDoomed += trueDoomed.size();
        if (curLost == opt) curOpt++; if (rwLost == opt) rwOpt++;
      }
    }
    System.out.printf(Locale.ROOT, "%s: %d instances (%d skipped); Lost CURRENT %d, RW_VITAL %d, RW_LOCAL %d, best of the two %d, optimum %d, worst %d; at the optimum: CURRENT %d, RW_VITAL %d, RW_LOCAL %d, best of the two %d; doomed: bound %d, exact %d, bound claims not doomed in truth %d%n",
      config, instances, skipped, sumCur, sumRw, sumLocal, sumBoth, sumOpt, sumWorst, curOpt, rwOpt, localOpt, bothOpt, sumLb, sumTrueDoomed, lbSoundFail);
    rows.close();
  }
  /** Positions carrying a node of P's Nettree that lies on a complete occurrence (root level to leaf level). */
  static TreeSet<Integer> onOccurrences(Pattern p, EventSequence seq, GapConstraint gap) {
    Nettree t = Nettree.build(p, seq, gap); int n = t.nodeCount();
    boolean[] fromRoot = new boolean[n], toLeaf = new boolean[n];
    // Nodes are numbered level by level, so one pass in each direction settles reachability.
    for (int v = 0; v < n; v++) { if (t.isSource(v)) fromRoot[v] = true; if (fromRoot[v]) for (int u : t.out(v)) fromRoot[u] = true; }
    for (int v = n - 1; v >= 0; v--) { if (t.isSink(v)) toLeaf[v] = true; if (toLeaf[v]) for (int u : t.in(v)) toLeaf[u] = true; }
    TreeSet<Integer> out = new TreeSet<>(); for (int v = 0; v < n; v++) if (fromRoot[v] && toLeaf[v]) out.add(t.position(v));
    return out;
  }
  /**
   * Marginal damage at the current solution, per masked sequence and watched pattern present there:
   * outside the masks, 1 on positions carrying a vital node of Q in the masked sequence; on a mask,
   * the support of Q regained by unmasking that position alone.
   */
  static Map<Integer, Map<Pattern, double[]>> marginal(SequenceDatabase db, GapConstraint gap, Map<Integer, Set<Integer>> masked, Map<Integer, Map<Pattern, Integer>> supIn) {
    Map<Integer, Map<Pattern, double[]>> out = new HashMap<>();
    for (Map.Entry<Integer, Set<Integer>> e : masked.entrySet()) {
      if (e.getValue().isEmpty()) continue;
      EventSequence orig = db.get(e.getKey()), after = orig.mask(e.getValue());
      NettreeCutIndex idx = new NettreeCutIndex(after, gap);
      Map<Pattern, double[]> m = new HashMap<>();
      for (Pattern q : supIn.get(e.getKey()).keySet()) {
        double[] a = new double[orig.length()];
        int id = idx.add(q, false); int now = idx.supNO(id);
        for (int pos : idx.vitalPositions(id)) a[pos] = 1;
        for (int i : e.getValue()) { Set<Integer> rest = new HashSet<>(e.getValue()); rest.remove(i); a[i] = SupportNO.of(q, orig.mask(rest), gap) - now; }
        m.put(q, a);
      }
      out.put(e.getKey(), m);
    }
    return out;
  }
  static long binomial(int n, int k) { long b = 1; for (int i = 0; i < k; i++) { b = b * (n - i) / (i + 1); if (b > Long.MAX_VALUE / 64) return Long.MAX_VALUE / 64; } return b; }
  static Map<Pattern, Long> drops(SequenceDatabase db, GapConstraint gap, Map<Integer, Set<Integer>> masked, Map<Integer, Map<Pattern, Integer>> supIn) {
    Map<Pattern, Long> drop = new HashMap<>();
    for (Map.Entry<Integer, Set<Integer>> e : masked.entrySet()) {
      if (e.getValue().isEmpty()) continue;
      EventSequence after = db.get(e.getKey()).mask(e.getValue());
      for (Map.Entry<Pattern, Integer> sq : supIn.get(e.getKey()).entrySet()) { int now = SupportNO.of(sq.getKey(), after, gap); if (now != sq.getValue()) drop.merge(sq.getKey(), (long) (sq.getValue() - now), Long::sum); }
    }
    return drop;
  }
