import hidenosp.core.*; import hidenosp.io.*; import hidenosp.metrics.*; import hidenosp.hide.*; import hidenosp.exp.SensitiveSampling; import hidenosp.index.NettreeCutIndex; import java.util.*; import java.nio.file.*;
/**
 * Probe: on one sensitive set, the exact Pareto front between |R| and the surrogate weight, once with
 * the weight of the method as it stands (CURRENT) and once with the weight RW_VITAL ends on (the
 * weight of its best iteration), and whether a vertex of either front dominates a given point
 * (|R|, Lost) of another method.
 */
public class FrontProbe {
  static final int ITERATIONS = 6;
  public static void main(String[] a) {
    String file = a[0]; int sigma = Integer.parseInt(a[1]); GapConstraint gap = GapConstraint.of(0, Integer.parseInt(a[2])); boolean repeated = a[3].equals("repeated");
    int[] items = Arrays.stream(a[4].split(" ")).mapToInt(Integer::parseInt).toArray(); Pattern target = new Pattern(items);
    int otherMasks = Integer.parseInt(a[5]), otherLost = Integer.parseInt(a[6]);
    SequenceDatabase db = DatasetReader.read(Paths.get(file), DatasetReader.Format.SPMF).database();
    Metrics.Context context = Metrics.Context.of(db, gap, 4, sigma);
    Map<Pattern, Long> f = context.frequent();
    Pattern p = null; for (SensitiveSampling.Draw d : SensitiveSampling.draw(f, 1, 4, 2026, repeated).draws()) if (d.patterns().get(0).equals(target)) p = target;
    if (p == null) throw new IllegalStateException("set " + target + " not in the sample");
    HideProblem problem = new HideProblem(db, List.of(p), gap, HideProblem.Mode.HIDE_DB, sigma, 0.5, 4);
    Set<Pattern> closure = FrequentPatterns.closure(List.of(p), f.keySet());
    List<Pattern> watched = new ArrayList<>(); for (Pattern q : f.keySet()) if (!closure.contains(q)) watched.add(q);
    List<Pattern> band = FrequentPatterns.nearThreshold(f, sigma, 0.5, closure);
    Metrics.Context.Session session = context.session(problem);
    int N = db.size();
    Map<Integer, Map<Pattern, int[]>> vital = new HashMap<>(); Map<Integer, Map<Pattern, Integer>> supIn = new HashMap<>();
    for (int s = 0; s < N; s++) {
      if (SupportNO.of(p, db.get(s), gap) == 0) continue;
      NettreeCutIndex idx = new NettreeCutIndex(db.get(s), gap);
      Map<Pattern, int[]> vs = new HashMap<>(); Map<Pattern, Integer> su = new HashMap<>();
      for (Pattern q : watched) { int id = idx.add(q, false); int nu = idx.supNO(id); if (nu == 0) continue; su.put(q, nu); vs.put(q, idx.vitalPositions(id)); }
      vital.put(s, vs); supIn.put(s, su);
    }
    Map<Pattern, Double> factor = new HashMap<>(); for (Pattern q : watched) factor.put(q, 1.0 / (f.get(q) - sigma + 1));
    double[][] bestW = null; int bestLost = Integer.MAX_VALUE; double[][] currentW = null;
    for (int it = 0; it < ITERATIONS; it++) {
      double[][] w = new double[N][];
      for (int s = 0; s < N; s++) {
        w[s] = new double[db.get(s).length()];
        if (!vital.containsKey(s)) continue;
        for (Pattern q : it == 0 ? band : watched) { int[] vp = vital.get(s).get(q); if (vp == null) continue; double c = it == 0 ? 1.0 / (f.get(q) - sigma + 1) : factor.get(q); for (int pos : vp) w[s][pos] += c; }
      }
      if (it == 0) currentW = w;
      Sanitization san = new CutHide(CutHideConfig.uniformWeights()).withExplicitWeights(w).run(problem);
      int lost = session.evaluate(san).lost();
      if (lost < bestLost) { bestLost = lost; bestW = w; }
      if (it + 1 < ITERATIONS) {
        Map<Pattern, Long> drop = new HashMap<>();
        for (Map.Entry<Integer, Set<Integer>> e : san.masked().entrySet()) {
          if (e.getValue().isEmpty()) continue;
          EventSequence after = db.get(e.getKey()).mask(e.getValue());
          for (Map.Entry<Pattern, Integer> sq : supIn.get(e.getKey()).entrySet()) { int now = SupportNO.of(sq.getKey(), after, gap); if (now != sq.getValue()) drop.merge(sq.getKey(), (long) (sq.getValue() - now), Long::sum); }
        }
        for (Pattern q : watched) { double m = f.get(q) - sigma + 1, dq = drop.getOrDefault(q, 0L), tau = Math.max(1.0, 0.25 * m); double sg = 1.0 / (1.0 + Math.exp(-(dq - m + 0.5) / tau)); factor.put(q, 1.0 / m + sg * (1 - sg) / tau); }
      }
    }
    StringBuilder out = new StringBuilder(String.format(Locale.ROOT, "%s sigma=%d %s %s set %s | other method |R| %d Lost %d | RW_VITAL at |R|* Lost %d |", Paths.get(file).getFileName(), sigma, gap, a[3], target, otherMasks, otherLost, bestLost));
    String[] names = {"front with CURRENT weight", "front with RW_VITAL weight"};
    double[][][] ws = {currentW, bestW};
    for (int k = 0; k < 2; k++) {
      List<CutHide.ParetoVertex> front = new CutHide(CutHideConfig.uniformWeights()).withExplicitWeights(ws[k]).paretoFront(problem);
      int bestWithin = Integer.MAX_VALUE; boolean dominates = false; int minM = Integer.MAX_VALUE, maxM = 0;
      for (CutHide.ParetoVertex v : front) {
        int lost = session.evaluate(v.sanitization()).lost(); int m = v.masks();
        minM = Math.min(minM, m); maxM = Math.max(maxM, m);
        if (m <= otherMasks) bestWithin = Math.min(bestWithin, lost);
        if (m <= otherMasks && lost <= otherLost && (m < otherMasks || lost < otherLost)) dominates = true;
      }
      out.append(String.format(Locale.ROOT, " %s: %d vertices, |R| %d..%d, least Lost within |R| <= %d: %s, dominates: %s |", names[k], front.size(), minM, maxM, otherMasks, bestWithin == Integer.MAX_VALUE ? "none" : String.valueOf(bestWithin), dominates ? "yes" : "no"));
    }
    System.out.println(out);
  }
}
