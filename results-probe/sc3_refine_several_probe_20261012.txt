## refcmp.py (repository path written as <repository>; arguments: the probe directories)
"""Compare the probe rows of cut-hide-refine with the official SC3 rows (several patterns) and the SC3LB bounds.

    python refcmp.py PROBE_DIR... 

Joins on (dataset, gap, sigma, scope, |S|, sensitive set). Stops with status 1 when a probe set has no
official row or no bound, when a refined row is not hidden (hf > 0), uses more masks than the official
greedy row, or uses fewer masks than the conflict bound.
"""
import csv, glob, io, re, statistics, sys
from collections import defaultdict
from pathlib import Path

REPO = Path("<repository>")

def rows_of(path):
    with io.open(path, encoding="utf-8", newline="") as h:
        return list(csv.DictReader([l for l in h if not l.startswith("#")]))

def options_of(path):
    with io.open(path, encoding="utf-8", newline="") as h:
        for l in h:
            if l.startswith("# options"):
                return dict(re.findall(r"(\w+)=([^,}]+)", l))
    raise ValueError(path)

def stem(t):
    return Path(t.replace("\\", "/")).stem

def key(r, o):
    return (stem(r["dataset"]), r["gap"], r["sigma"], o["patternKind"], o["sensitiveSize"], r["sensitive_set"])

official = defaultdict(dict)
for p in sorted(glob.glob(str(REPO / "results-invariant" / "SC3_*DESKTOP-L8A6EPO*.csv"))):
    o = options_of(p)
    if o.get("methods") == "none" or o.get("sensitiveSize") == "1" or o.get("group", "main") != "main":
        continue
    for r in rows_of(p):
        official[key(r, o)][r["method_key"]] = r
bounds = {}
for p in sorted(glob.glob(str(REPO / "results-invariant" / "SC3LB_*.csv"))):
    for r in rows_of(p):
        bounds[(stem(r["dataset"]), r["gap"], r["sigma"], r["pattern_kind"], r["sensitive_size"], r["sensitive_set"])] = r
probe = {}
files = 0
for d in sys.argv[1:]:
    for p in sorted(glob.glob(d + "/*/results-probe/SC3_*.csv")):
        files += 1
        o = options_of(p)
        for r in rows_of(p):
            if r["method_key"] == "cut-hide-refine":
                probe[key(r, o)] = r

problems = []
print("probe files %d, probe sets %d, official sets (|S|>1, main) %d" % (files, len(probe), len(official)))
missing = [k for k in official if k not in probe]
print("official sets without a probe row: %d" % len(missing))
groups = defaultdict(list)
for k, r in probe.items():
    if k not in official or "cut-hide-rw-ls" not in official[k]:
        problems.append("no official row: %s" % (k,)); continue
    if k not in bounds:
        problems.append("no bound: %s" % (k,)); continue
    groups[(k[3], int(k[4]))].append(k)

for (scope, size), keys in sorted(groups.items()):
    stats = defaultdict(int)
    dm, dl, ro, rr, rh = [], [], [], [], []
    hha_better_before = hha_better_after = 0
    for k in keys:
        new, old, hha, b = probe[k], official[k]["cut-hide-rw-ls"], official[k].get("hha"), bounds[k]
        if new["status"] or old["status"]:
            stats["status"] += 1; continue
        stats["n"] += 1
        mn, mo = int(new["mask_count"]), int(old["mask_count"])
        lb = int(b["conflict_bound"])
        if float(new["hf"]) > 0: problems.append("not hidden: %s" % (k,))
        if mn > mo: problems.append("more masks %d > %d: %s" % (mn, mo, k))
        if mn < lb: problems.append("below bound %d < %d: %s" % (mn, lb, k))
        ln, lo = int(new["lost"]), int(old["lost"])
        stats["fewer"] += mn < mo
        stats["lost_up"] += ln > lo
        stats["lost_down"] += ln < lo
        stats["reach"] += mn == lb
        stats["reach_before"] += mo == lb
        dm.append((mo - mn) / mo if mo else 0.0)
        dl.append(ln - lo)
        if lb:
            ro.append(mo / lb); rr.append(mn / lb)
        if hha and not hha["status"]:
            mh = int(hha["mask_count"])
            hha_better_before += mh < mo
            hha_better_after += mh < mn
            if lb: rh.append(mh / lb)
    n = stats["n"]
    print("\n== %s, |S| = %d: %d sets (%d with a status)" % (scope, size, n, stats["status"]))
    print("   fewer masks than the greedy: %d/%d; relative saving median %.4f, max %.4f"
          % (stats["fewer"], n, statistics.median(dm), max(dm)))
    print("   Lost raised on %d/%d, lowered on %d/%d; change median %s, min %d, max %d"
          % (stats["lost_up"], n, stats["lost_down"], n, statistics.median(dl), min(dl), max(dl)))
    print("   |R| = bound: greedy %d/%d, refined %d/%d" % (stats["reach_before"], n, stats["reach"], n))
    print("   ratio to the bound median: greedy %.4f, refined %.4f, hha %.4f; max: greedy %.4f, refined %.4f"
          % (statistics.median(ro), statistics.median(rr), statistics.median(rh), max(ro), max(rr)))
    print("   sets where hha uses fewer masks: against the greedy %d, against the refined %d"
          % (hha_better_before, hha_better_after))
print()
if problems:
    print("PROBLEMS (%d):" % len(problems))
    for p in problems[:20]: print("  " + p)
    sys.exit(1)
print("No problem: every refined set is hidden, never above the greedy, never below the bound.")
