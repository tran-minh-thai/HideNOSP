===== harness source: SurrogateProbe.java =====
import hidenosp.core.*; import hidenosp.io.*; import hidenosp.metrics.*; import hidenosp.hide.*; import hidenosp.exp.SensitiveSampling; import hidenosp.index.NettreeCutIndex; import java.util.*; import java.nio.file.*; import java.io.*;
/**
 * Probe: which additive surrogate makes the exact CUT-HIDE lose the fewest frequent patterns?
 * Every candidate is solved masks-first, so every candidate masks exactly |R|* positions and only
 * the choice among minimum-mask solutions differs.
 *  CURRENT      the method as it stands: vital positions of B_eps (sup < 1.5 sigma), factor 1/m, m = sup - sigma + 1
 *  VITAL_BAND   the same weight rebuilt here, a check of this harness (must give the same |R| and Lost)
 *  VITAL_ALL    vital positions of every watched pattern, factor 1/m
 *  OCC_BAND     occurrence share nu*occ(i)/(|Q|*occ) of the band, factor 1/m
 *  OCC_ALL      occurrence share of every watched pattern, factor 1/m
 *  MIX_ALL      vital + occurrence share, every watched pattern, factor 1/m
 *  RW_VITAL     starts at CURRENT, then 5 re-weightings on the exact drops (vital basis), best Lost kept
 *  RW_OCC       starts at OCC_ALL, then 5 re-weightings on the exact drops (occurrence basis), best Lost kept
 * Re-weighting: factor(Q) = 1/m + s(1-s)/tau, s = logistic((d_Q - m + 0.5)/tau), tau = max(1, m/4),
 * d_Q the exact drop of Q under the previous solution (the derivative of a smoothed threshold).
 */
public class SurrogateProbe {
  static final String[] NAMES = {"CURRENT", "VITAL_BAND", "VITAL_ALL", "OCC_BAND", "OCC_ALL", "MIX_ALL", "RW_VITAL", "RW_OCC"};
  static final int ITERATIONS = 6;
  public static void main(String[] a) throws IOException {
    String file = a[0]; int sigma = Integer.parseInt(a[1]); GapConstraint gap = GapConstraint.of(0, Integer.parseInt(a[2])); boolean repeated = a[3].equals("repeated");
    PrintWriter rows = new PrintWriter(new FileWriter(a[4], true));
    String config = Paths.get(file).getFileName() + "," + sigma + "," + a[2] + "," + a[3];
    SequenceDatabase db = DatasetReader.read(Paths.get(file), DatasetReader.Format.SPMF).database();
    Metrics.Context context = Metrics.Context.of(db, gap, 4, sigma);
    Map<Pattern, Long> f = context.frequent();
    SensitiveSampling.Result sampling = SensitiveSampling.draw(f, 1, 4, 2026, repeated);
    int n = NAMES.length; int[] totalLost = new int[n]; int[] better = new int[n], worse = new int[n], equal = new int[n]; long[] nanos = new long[n];
    int sets = 0, harnessMismatch = 0, maskMismatch = 0;
    for (SensitiveSampling.Draw d : sampling.draws()) {
      Pattern p = d.patterns().get(0);
      HideProblem problem = new HideProblem(db, List.of(p), gap, HideProblem.Mode.HIDE_DB, sigma, 0.5, 4);
      Set<Pattern> closure = FrequentPatterns.closure(List.of(p), f.keySet());
      List<Pattern> watched = new ArrayList<>(); for (Pattern q : f.keySet()) if (!closure.contains(q)) watched.add(q);
      List<Pattern> band = FrequentPatterns.nearThreshold(f, sigma, 0.5, closure);
      Set<Pattern> bandSet = new HashSet<>(band);
      Metrics.Context.Session session = context.session(problem);
      int N = db.size();
      long t0 = System.nanoTime();
      // Per sequence carrying P: per watched pattern present, its support, vital positions and occurrence shares.
      Map<Integer, Map<Pattern, int[]>> vital = new HashMap<>(); Map<Integer, Map<Pattern, double[]>> share = new HashMap<>(); Map<Integer, Map<Pattern, Integer>> supIn = new HashMap<>();
      for (int s = 0; s < N; s++) {
        EventSequence seq = db.get(s);
        if (SupportNO.of(p, seq, gap) == 0) continue;
        NettreeCutIndex idx = new NettreeCutIndex(seq, gap);
        Map<Pattern, int[]> vs = new HashMap<>(); Map<Pattern, double[]> os = new HashMap<>(); Map<Pattern, Integer> su = new HashMap<>();
        for (Pattern q : watched) {
          int id = idx.add(q, false); int nu = idx.supNO(id); if (nu == 0) continue;
          su.put(q, nu); vs.put(q, idx.vitalPositions(id));
          long total = idx.totalOccurrences(id); double[] sh = new double[seq.length()];
          for (int i = 0; i < seq.length(); i++) { long occ = idx.occurrencesThrough(id, i); if (occ > 0) sh[i] = (double) nu * occ / ((double) q.length() * total); }
          os.put(q, sh);
        }
        vital.put(s, vs); share.put(s, os); supIn.put(s, su);
      }
      long prep = System.nanoTime() - t0;
      Map<Pattern, Double> base = new HashMap<>(); for (Pattern q : watched) base.put(q, 1.0 / (f.get(q) - sigma + 1));
      int[] lost = new int[n]; int[] masks = new int[n]; int[] bestIt = new int[n];
      for (int k = 0; k < n; k++) {
        long start = System.nanoTime();
        String name = NAMES[k];
        if (name.equals("CURRENT")) {
          double[] factor = FrequentPatterns.weightFactors(band, f, sigma);
          Sanitization san = new CutHide(CutHideConfig.standard()).withNearThreshold(band, factor).run(problem);
          Metrics.Row row = session.evaluate(san); lost[k] = row.lost(); masks[k] = san.maskCount();
          nanos[k] += System.nanoTime() - start;
          continue;
        }
        boolean all = name.endsWith("_ALL") || name.startsWith("RW");
        boolean occBasis = name.contains("OCC") || name.startsWith("MIX");
        boolean vitalBasis = name.contains("VITAL") || name.startsWith("MIX");
        Map<Pattern, Double> factor = new HashMap<>(base);
        boolean onlyBandFirst = name.equals("RW_VITAL");  // iteration 0 of RW_VITAL is CURRENT
        int iterations = name.startsWith("RW") ? ITERATIONS : 1;
        int best = Integer.MAX_VALUE; int bestMasks = -1; int arg = -1;
        for (int it = 0; it < iterations; it++) {
          double[][] w = new double[N][];
          for (int s = 0; s < N; s++) {
            w[s] = new double[db.get(s).length()];
            if (!vital.containsKey(s)) continue;
            // Sum in the order of the band list first, so that VITAL_BAND reproduces CURRENT bit for bit.
            List<Pattern> order = (!all || (onlyBandFirst && it == 0)) ? band : watched;
            for (Pattern q : order) {
              int[] vp = vital.get(s).get(q); if (vp == null) continue;
              double c = (onlyBandFirst && it == 0) || !name.startsWith("RW") ? base.get(q) : factor.get(q);
              if (vitalBasis) for (int pos : vp) w[s][pos] += c;
              if (occBasis) { double[] sh = share.get(s).get(q); for (int i = 0; i < sh.length; i++) if (sh[i] != 0) w[s][i] += c * sh[i]; }
            }
          }
          Sanitization san = new CutHide(CutHideConfig.uniformWeights()).withExplicitWeights(w).run(problem);
          Metrics.Row row = session.evaluate(san);
          if (row.lost() < best) { best = row.lost(); bestMasks = san.maskCount(); arg = it; }
          if (bestMasks != san.maskCount()) maskMismatch++;
          if (it + 1 < iterations) {
            Map<Pattern, Long> drop = new HashMap<>();
            for (Map.Entry<Integer, Set<Integer>> e : san.masked().entrySet()) {
              if (e.getValue().isEmpty()) continue;
              EventSequence after = db.get(e.getKey()).mask(e.getValue());
              for (Map.Entry<Pattern, Integer> sq : supIn.get(e.getKey()).entrySet()) {
                int now = SupportNO.of(sq.getKey(), after, gap);
                if (now != sq.getValue()) drop.merge(sq.getKey(), (long) (sq.getValue() - now), Long::sum);
              }
            }
            for (Pattern q : watched) {
              double m = f.get(q) - sigma + 1, dq = drop.getOrDefault(q, 0L), tau = Math.max(1.0, 0.25 * m);
              double sg = 1.0 / (1.0 + Math.exp(-(dq - m + 0.5) / tau));
              factor.put(q, 1.0 / m + sg * (1 - sg) / tau);
            }
          }
        }
        lost[k] = best; masks[k] = bestMasks; bestIt[k] = arg;
        nanos[k] += System.nanoTime() - start;
      }
      if (lost[1] != lost[0] || masks[1] != masks[0]) harnessMismatch++;
      for (int k = 0; k < n; k++) {
        if (masks[k] != masks[0]) maskMismatch++;
        totalLost[k] += lost[k];
        if (lost[k] < lost[0]) better[k]++; else if (lost[k] > lost[0]) worse[k]++; else equal[k]++;
        rows.printf(Locale.ROOT, "%s,%d,%s,%s,%d,%d,%d,%d%n", config, sets, p, NAMES[k], masks[k], lost[k], bestIt[k], watched.size());
      }
      rows.flush();
      sets++;
      nanos[0] += 0; if (sets == 1) System.err.printf(Locale.ROOT, "first set: preparation %.1f s%n", prep / 1e9);
    }
    StringBuilder b = new StringBuilder(String.format(Locale.ROOT, "%s sigma=%d %s %s: %d sets; harness check VITAL_BAND vs CURRENT mismatches %d; |R| mismatches %d |", Paths.get(file).getFileName(), sigma, gap, a[3], sets, harnessMismatch, maskMismatch));
    for (int k = 0; k < n; k++) b.append(String.format(Locale.ROOT, " %s lost %d (vs CURRENT: %d fewer / %d equal / %d more; %.1f s) |", NAMES[k], totalLost[k], better[k], equal[k], worse[k], nanos[k] / 1e9));
    System.out.println(b);
    rows.close();
  }
}

"""Summarise the surrogate probe rows: per configuration, total Lost per candidate and the count of
sets where it loses fewer / the same / more patterns than the current weight. Numbers come only
from the row files."""
import csv, glob, os, sys, collections
out = sys.argv[1]
header = open(os.path.join(out, "rows_header.csv")).read().strip().split(",")
rows = []
for f in sorted(glob.glob(os.path.join(out, "rows_*.csv"))):
    if f.endswith("rows_header.csv"):
        continue
    for line in open(f):
        parts = line.rstrip("\n").split(",")
        # The pattern column holds commas ("[9, 9, 2]"): five fixed fields before it, five after.
        if len(parts) < 11 or not parts[-5] in ("CURRENT", "VITAL_BAND", "VITAL_ALL", "OCC_BAND", "OCC_ALL", "MIX_ALL", "RW_VITAL", "RW_OCC"):
            raise SystemExit("malformed row in %s: %s" % (f, line))
        rows.append(dict(zip(header, parts[:5] + [",".join(parts[5:-5])] + parts[-5:])))
by = collections.defaultdict(dict)
for r in rows:
    by[(r["dataset"], r["sigma"], r["gmax"], r["kind"], r["set"])][r["candidate"]] = r
cands = ["CURRENT", "VITAL_BAND", "VITAL_ALL", "OCC_BAND", "OCC_ALL", "MIX_ALL", "RW_VITAL", "RW_OCC"]
configs = sorted({k[:4] for k in by}, key=lambda c: (c[3], c[0]))
print("| data | sigma | gap | kind | sets | " + " | ".join(cands) + " |")
print("|---|---:|---|---|---:|" + "---:|" * len(cands))
grand = collections.Counter(); cmp = collections.defaultdict(collections.Counter); problems = 0
for c in configs:
    sets = [k for k in by if k[:4] == c]
    tot = collections.Counter()
    for k in sets:
        cur = by[k]["CURRENT"]
        for cand in cands:
            r = by[k][cand]
            if r["masks"] != cur["masks"]:
                problems += 1
            tot[cand] += int(r["lost"])
            d = int(r["lost"]) - int(cur["lost"])
            cmp[cand]["fewer" if d < 0 else "more" if d > 0 else "same"] += 1
    for cand in cands:
        grand[cand] += tot[cand]
    print("| %s | %s | [0,%s] | %s | %d | " % (c[0].replace(".txt", ""), c[1], c[2], c[3], len(sets))
          + " | ".join(str(tot[x]) for x in cands) + " |")
print("| all | | | | %d | " % len(by) + " | ".join(str(grand[x]) for x in cands) + " |")
print()
for cand in cands:
    print("%s: against CURRENT over %d sets: %d fewer lost, %d same, %d more"
          % (cand, len(by), cmp[cand]["fewer"], cmp[cand]["same"], cmp[cand]["more"]))
print("sets where a candidate masks a different number of positions than CURRENT: %d" % problems)
